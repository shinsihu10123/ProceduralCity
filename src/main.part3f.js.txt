function dot3(a, b) {
  return a[0] * b[0] + a[1] * b[1] + a[2] * b[2]
}

function mat4LookAt(out, eye, target, up) {
  const zAxis = normalize3([eye[0] - target[0], eye[1] - target[1], eye[2] - target[2]])
  const xAxis = normalize3(cross3(up, zAxis))
  const yAxis = cross3(zAxis, xAxis)
  out[0] = xAxis[0]; out[1] = yAxis[0]; out[2] = zAxis[0]; out[3] = 0
  out[4] = xAxis[1]; out[5] = yAxis[1]; out[6] = zAxis[1]; out[7] = 0
  out[8] = xAxis[2]; out[9] = yAxis[2]; out[10] = zAxis[2]; out[11] = 0
  out[12] = -dot3(xAxis, eye); out[13] = -dot3(yAxis, eye); out[14] = -dot3(zAxis, eye); out[15] = 1
  return out
}

function mat4Multiply(out, a, b) {
  const result = new Float32Array(16)
  for (let column = 0; column < 4; column += 1) {
    for (let row = 0; row < 4; row += 1) {
      result[column * 4 + row] =
        a[row] * b[column * 4]
        + a[4 + row] * b[column * 4 + 1]
        + a[8 + row] * b[column * 4 + 2]
        + a[12 + row] * b[column * 4 + 3]
    }
  }
  out.set(result)
  return out
}

function v053NodeKey(x, z) {
  return `${Math.round(x * 10) / 10}:${Math.round(z * 10) / 10}`
}

function v053RoadRange(road) {
  const half = Math.max(road.length || 0, 8) / 2
  return road.orientation === 'horizontal'
    ? { minX: road.x - half, maxX: road.x + half, minZ: road.z, maxZ: road.z }
    : { minX: road.x, maxX: road.x, minZ: road.z - half, maxZ: road.z + half }
}

function v053BuildRoadGraph(city) {
  const roads = (city?.roads || []).filter((road) => road.length > 8)
  const nodeMap = new Map()
  const roadPoints = new Map()

  function ensureNode(x, y, z) {
    const key = v053NodeKey(x, z)
    if (!nodeMap.has(key)) nodeMap.set(key, { id: key, x, y, z, edges: [] })
    return nodeMap.get(key)
  }

  function endpoints(road) {
    if (road.start && road.end) return [road.start, road.end]
    const range = v053RoadRange(road)
    return road.orientation === 'horizontal'
      ? [{ x: range.minX, y: road.y || 0, z: road.z }, { x: range.maxX, y: road.y || 0, z: road.z }]
      : [{ x: road.x, y: road.y || 0, z: range.minZ }, { x: road.x, y: road.y || 0, z: range.maxZ }]
  }

  function intersection(a0, a1, b0, b1) {
    const adx = a1.x - a0.x
    const adz = a1.z - a0.z
    const bdx = b1.x - b0.x
    const bdz = b1.z - b0.z
    const denominator = adx * bdz - adz * bdx
    if (Math.abs(denominator) < 1e-7) return null
    const ox = b0.x - a0.x
    const oz = b0.z - a0.z
    const ta = (ox * bdz - oz * bdx) / denominator
    const tb = (ox * adz - oz * adx) / denominator
    if (ta < -0.001 || ta > 1.001 || tb < -0.001 || tb > 1.001) return null
    return {
      x: a0.x + adx * ta,
      z: a0.z + adz * ta,
      ta: clamp(ta, 0, 1),
      tb: clamp(tb, 0, 1),
    }
  }

  roads.forEach((road, index) => {
    const [start, end] = endpoints(road)
    roadPoints.set(index, [{ ...start, t: 0 }, { ...end, t: 1 }])
  })

  for (let a = 0; a < roads.length; a += 1) {
    const [a0, a1] = endpoints(roads[a])
    const minAX = Math.min(a0.x, a1.x) - 0.25
    const maxAX = Math.max(a0.x, a1.x) + 0.25
    const minAZ = Math.min(a0.z, a1.z) - 0.25
    const maxAZ = Math.max(a0.z, a1.z) + 0.25
    for (let b = a + 1; b < roads.length; b += 1) {
      const [b0, b1] = endpoints(roads[b])
      if (Math.max(b0.x, b1.x) < minAX || Math.min(b0.x, b1.x) > maxAX || Math.max(b0.z, b1.z) < minAZ || Math.min(b0.z, b1.z) > maxAZ) continue
      const point = intersection(a0, a1, b0, b1)
      if (!point) continue
      roadPoints.get(a).push({ x: point.x, z: point.z, y: a0.y + (a1.y - a0.y) * point.ta, t: point.ta })
      roadPoints.get(b).push({ x: point.x, z: point.z, y: b0.y + (b1.y - b0.y) * point.tb, t: point.tb })
    }
  }

  const edges = []
  roads.forEach((road, roadIndex) => {
    const points = roadPoints.get(roadIndex)
    points.sort((a, b) => a.t - b.t)
    const unique = []
    for (const point of points) {
      if (!unique.length || Math.hypot(point.x - unique[unique.length - 1].x, point.z - unique[unique.length - 1].z) > 0.8) unique.push(point)
    }
    for (let index = 0; index < unique.length - 1; index += 1) {
      const startPoint = unique[index]
      const endPoint = unique[index + 1]
      const length = Math.hypot(endPoint.x - startPoint.x, endPoint.z - startPoint.z)
      if (length < 4) continue
      const start = ensureNode(startPoint.x, startPoint.y ?? road.y ?? 0, startPoint.z)
      const end = ensureNode(endPoint.x, endPoint.y ?? road.y ?? 0, endPoint.z)
      const edge = {
        id: `E-${edges.length + 1}`,
        start,
        end,
        length,
        roadClass: road.class || 'local',
        orientation: road.orientation,
      }
      const reverse = {
        id: `${edge.id}R`,
        start: end,
        end: start,
        length,
        roadClass: edge.roadClass,
        orientation: edge.orientation,
      }
      start.edges.push(edge)
      end.edges.push(reverse)
      edges.push(edge, reverse)
    }
  })

  return { nodes: [...nodeMap.values()], edges }
}

function v053Heuristic(a, b) {
  return Math.hypot(a.x - b.x, a.z - b.z)
}

function v053FindRoute(graph, start, goal) {
  if (!start || !goal || start === goal) return []
  const open = [start]
  const cameFrom = new Map()
  const edgeFrom = new Map()
  const gScore = new Map([[start.id, 0]])
  const fScore = new Map([[start.id, v053Heuristic(start, goal)]])

  while (open.length) {
    open.sort((a, b) => (fScore.get(a.id) ?? Infinity) - (fScore.get(b.id) ?? Infinity))
    const current = open.shift()
    if (current === goal) {
      const route = []
      let cursor = current
      while (cameFrom.has(cursor.id)) {
        route.unshift(edgeFrom.get(cursor.id))
        cursor = cameFrom.get(cursor.id)
      }
      return route
    }
    for (const edge of current.edges) {
      const congestionPenalty = edge.roadClass === 'local' ? 1.18 : edge.roadClass === 'arterial' ? 0.88 : 1
      const tentative = (gScore.get(current.id) ?? Infinity) + edge.length * congestionPenalty
      if (tentative >= (gScore.get(edge.end.id) ?? Infinity)) continue
      cameFrom.set(edge.end.id, current)
      edgeFrom.set(edge.end.id, edge)
      gScore.set(edge.end.id, tentative)
      fScore.set(edge.end.id, tentative + v053Heuristic(edge.end, goal))
      if (!open.includes(edge.end)) open.push(edge.end)
    }
  }
  return []
}

function v053ChooseDestination(graph, origin, index) {
  if (!graph.nodes.length) return origin
  for (let attempt = 0; attempt < 12; attempt += 1) {
    const candidate = graph.nodes[(index * 17 + attempt * 31 + Math.floor(performance.now() / 5000)) % graph.nodes.length]
    if (candidate !== origin && v053Heuristic(origin, candidate) > 80) return candidate
  }
  return graph.nodes[(index + Math.floor(graph.nodes.length / 2)) % graph.nodes.length]
}

function v053AssignRoute(agent, index) {
  if (!v053Graph?.nodes.length) return
  const origin = agent.edge?.end || v053Graph.nodes[(index * 11) % v053Graph.nodes.length]
  const destination = v053ChooseDestination(v053Graph, origin, index)
  const route = v053FindRoute(v053Graph, origin, destination)
  if (route.length) {
    agent.route = route
    agent.routeIndex = 0
    agent.edge = route[0]
    agent.progress = 0
    agent.destination = destination
  } else if (origin.edges.length) {
    agent.route = [origin.edges[index % origin.edges.length]]
    agent.routeIndex = 0
    agent.edge = agent.route[0]
    agent.progress = 0
  }
}

function v053CreateRailRoute(city) {
  let points = city?.infrastructure?.transitLines?.[0]?.points
  if (!Array.isArray(points) || points.length < 2) {
    const extent = Math.max(city?.extent || 600, 160)
    const x = city?.centers?.cbd?.x || 0
    const y = city?.infrastructure?.stations?.[0]?.y ?? 1.8
    points = [{ x, y, z: -extent * 0.47 }, { x, y, z: extent * 0.47 }]
  }
  const cumulative = [0]
  for (let index = 0; index < points.length - 1; index += 1) {
    cumulative.push(cumulative[index] + Math.hypot(points[index + 1].x - points[index].x, points[index + 1].z - points[index].z))
  }
  return { points, cumulative, length: Math.max(1, cumulative[cumulative.length - 1]) }
}

function v053RailProgressForPoint(route, target) {
  let best = { distance: Infinity, progress: 0.5 }
  for (let index = 0; index < route.points.length - 1; index += 1) {
    const a = route.points[index]
    const b = route.points[index + 1]
    const dx = b.x - a.x
    const dz = b.z - a.z
    const length2 = dx * dx + dz * dz || 1
    const t = clamp(((target.x - a.x) * dx + (target.z - a.z) * dz) / length2, 0, 1)
    const x = a.x + dx * t
    const z = a.z + dz * t
    const candidate = Math.hypot(target.x - x, target.z - z)
    if (candidate < best.distance) {
      const segmentLength = Math.sqrt(length2)
      best = { distance: candidate, progress: (route.cumulative[index] + segmentLength * t) / route.length }
    }
  }
  return clamp(best.progress, 0.04, 0.96)
}

function v053RailPointAt(route, progress) {
  const targetDistance = clamp(progress, 0, 1) * route.length
  for (let index = 0; index < route.points.length - 1; index += 1) {
    const endDistance = route.cumulative[index + 1]
    if (targetDistance > endDistance && index < route.points.length - 2) continue
    const a = route.points[index]
    const b = route.points[index + 1]
    const segmentLength = Math.max(0.001, endDistance - route.cumulative[index])
    const t = clamp((targetDistance - route.cumulative[index]) / segmentLength, 0, 1)
    return {
      x: a.x + (b.x - a.x) * t,
      y: (a.y || 0) + ((b.y || 0) - (a.y || 0)) * t,
      z: a.z + (b.z - a.z) * t,
      dx: b.x - a.x,
      dz: b.z - a.z,
    }
  }
  const last = route.points[route.points.length - 1]
  return { ...last, dx: 0, dz: 1 }
}

function v053ResetTraffic(city) {
  v053TrafficCity = city
  v053Graph = v053BuildRoadGraph(city)
  v053TrafficAgents = []
  v053TrafficVertexCount = 0
  if (!v053Graph.nodes.length || !v053Graph.edges.length) return

  const density = Number(ui.density?.value || 68) / 100
  const count = Math.min(42, Math.max(14, Math.round(v053Graph.nodes.length * 0.12 * density)))
  for (let index = 0; index < count; index += 1) {
    const type = index % 11 === 0 ? 'bus' : index % 8 === 0 ? 'truck' : 'car'
    const speed = type === 'bus' ? 8.2 : type === 'truck' ? 7.0 : 10.5 + (index % 5) * 0.9
    const startNode = v053Graph.nodes[(index * 13) % v053Graph.nodes.length]
    const agent = {
      id: `V053-${index + 1}`,
      type,
      speed,
      color: v053TrafficColors[index % v053TrafficColors.length],
      laneSign: index % 2 === 0 ? 1 : -1,
      edge: null,
      route: [],
      routeIndex: 0,
      progress: 0,
      dwell: index * 0.08,
      stopsPassed: 0,
    }
    agent.edge = startNode.edges[index % Math.max(startNode.edges.length, 1)] || null
    v053AssignRoute(agent, index)
    agent.progress = ((index * 0.137) % 0.82) + 0.08
    v053TrafficAgents.push(agent)
  }

  const stations = city?.infrastructure?.stations || []
  const route = v053CreateRailRoute(city)
  const stationProgress = stations.length >= 2
    ? stations.map((station) => v053RailProgressForPoint(route, station)).sort((a, b) => a - b)
    : [0.14, 0.36, 0.58, 0.82]
  v053Rail = {
    progress: 0.10,
    direction: 1,
    dwell: 0,
    stations: stationProgress,
    routeLength: route.length,
    route,
  }

  window.dispatchEvent(new CustomEvent('procedural-city:webgl-traffic-ready', {
    detail: {
      vehicles: v053TrafficAgents.length,
      nodes: v053Graph.nodes.length,
      edges: v053Graph.edges.length,
      routed: true,
    },
  }))
}

function v053PushWheel(vertices, x, y, z, rotation, across, along, scale = 1) {
  const wx = x + Math.cos(rotation) * across - Math.sin(rotation) * along
  const wz = z + Math.sin(rotation) * across + Math.cos(rotation) * along
  pushBox(vertices, wx, y, wz, 0.38 * scale, 0.54 * scale, 0.72 * scale, [0.025, 0.03, 0.035], rotation)
}

function v053PushVehicle(vertices, agent, x, y, z, rotation) {
  const isBus = agent.type === 'bus'
  const isTruck = agent.type === 'truck'
  const scale = 1.22
  const length = (isBus ? 10.8 : isTruck ? 8.4 : 4.9) * scale
  const width = (isBus ? 2.52 : isTruck ? 2.42 : 1.96) * scale
  const bodyHeight = (isBus ? 1.68 : isTruck ? 1.52 : 0.80) * scale
  const cabinHeight = (isBus ? 1.12 : isTruck ? 1.08 : 0.72) * scale

  pushBox(vertices, x, y, z, width, bodyHeight, length, agent.color, rotation)
  if (isBus) {
    pushBox(vertices, x, y + 1.42 * scale, z, width * 0.93, cabinHeight, length * 0.91, [0.10, 0.24, 0.32], rotation)
    pushBox(vertices, x, y + 2.20 * scale, z, width * 0.74, 0.18, length * 0.64, [0.78, 0.82, 0.84], rotation)
  } else if (isTruck) {
    const forward = length * 0.31
    const dx = -Math.sin(rotation) * forward
    const dz = Math.cos(rotation) * forward
    pushBox(vertices, x + dx, y + 1.20 * scale, z + dz, width * 0.92, cabinHeight, length * 0.31, [0.11, 0.25, 0.33], rotation)
    pushBox(vertices, x - dx * 0.42, y + 1.08 * scale, z - dz * 0.42, width * 0.95, cabinHeight * 1.06, length * 0.57, agent.color, rotation)
  } else {
    pushBox(vertices, x, y + 0.84 * scale, z, width * 0.84, cabinHeight, length * 0.54, [0.10, 0.25, 0.34], rotation)
  }

  const wheelAlong = length * 0.31
  const wheelAcross = width * 0.53
  v053PushWheel(vertices, x, y - 0.34, z, rotation, wheelAcross, wheelAlong, scale)
  v053PushWheel(vertices, x, y - 0.34, z, rotation, -wheelAcross, wheelAlong, scale)
  v053PushWheel(vertices, x, y - 0.34, z, rotation, wheelAcross, -wheelAlong, scale)
  v053PushWheel(vertices, x, y - 0.34, z, rotation, -wheelAcross, -wheelAlong, scale)

  const lampOffset = length * 0.51
  pushBox(
    vertices,
    x - Math.sin(rotation) * lampOffset,
    y + bodyHeight * 0.12,
    z + Math.cos(rotation) * lampOffset,
    width * 0.62,
    0.14,
    0.10,
    [1.0, 0.88, 0.44],
    rotation,
  )
}

function v053AdvanceAgent(agent, deltaSeconds, index) {
  if (!agent.edge) {
    v053AssignRoute(agent, index)
    return
  }
  if (agent.dwell > 0) {
    agent.dwell = Math.max(0, agent.dwell - deltaSeconds)
    return
  }
  agent.progress += deltaSeconds * agent.speed / Math.max(agent.edge.length, 1)
  while (agent.progress >= 1) {
    agent.progress -= 1
    agent.routeIndex += 1
    agent.stopsPassed += 1
    if (agent.type === 'bus' && agent.stopsPassed % 3 === 0) agent.dwell = 1.6
    if (agent.routeIndex >= agent.route.length) {
      v053AssignRoute(agent, index + agent.stopsPassed)
      if (!agent.edge) return
    } else {
      agent.edge = agent.route[agent.routeIndex]
    }
  }
}

function v053BuildTraffic(city, deltaSeconds = 0) {
  if (city !== v053TrafficCity) v053ResetTraffic(city)
  if (!v053TrafficAgents.length) {
    v053TrafficVertexCount = 0
    return
  }

  const vertices = []
  v053TrafficAgents.forEach((agent, index) => {
    v053AdvanceAgent(agent, deltaSeconds, index)
    const edge = agent.edge
    if (!edge) return
    const t = clamp(agent.progress, 0, 1)
    const centerX = edge.start.x + (edge.end.x - edge.start.x) * t
    const centerZ = edge.start.z + (edge.end.z - edge.start.z) * t
    const y = edge.start.y + (edge.end.y - edge.start.y) * t + 0.62
    const dx = edge.end.x - edge.start.x
    const dz = edge.end.z - edge.start.z
    const rotation = Math.atan2(-dx, dz)
    const laneWidth = edge.roadClass === 'arterial' ? 4.8 : edge.roadClass === 'local' ? 1.55 : 2.65
    const laneX = Math.cos(rotation) * laneWidth * agent.laneSign
    const laneZ = Math.sin(rotation) * laneWidth * agent.laneSign
    v053PushVehicle(vertices, agent, centerX + laneX, y, centerZ + laneZ, rotation)
  })

  const data = new Float32Array(vertices)
  gl.bindBuffer(gl.ARRAY_BUFFER, v053TrafficBuffer)
  gl.bufferData(gl.ARRAY_BUFFER, data, gl.DYNAMIC_DRAW)
  v053TrafficVertexCount = data.length / 9
}

function v053CrossedStation(previous, current, station, direction) {
  return direction > 0
    ? previous < station && current >= station
    : previous > station && current <= station
}

function v053PushRailCar(vertices, x, y, z, rotation, lead, direction) {
  const color = [0.10, 0.68, 0.86]
  pushBox(vertices, x, y, z, 3.48, 2.55, 14.2, color, rotation)
  pushBox(vertices, x, y + 1.66, z, 3.30, 1.02, 12.4, [0.08, 0.24, 0.32], rotation)
  pushBox(vertices, x, y + 2.78, z, 2.42, 0.18, 8.0, [0.76, 0.82, 0.84], rotation)
  if (lead) {
    const offset = 7.18 * direction
    pushBox(vertices, x - Math.sin(rotation) * offset, y + 0.36, z + Math.cos(rotation) * offset, 2.1, 0.20, 0.12, [1.0, 0.86, 0.40], rotation)
  }
}

function v053BuildRail(city, deltaSeconds) {
  if (!v053Rail) {
    v053RailVertexCount = 0
    return
  }
  if (v053Rail.dwell > 0) {
    v053Rail.dwell = Math.max(0, v053Rail.dwell - deltaSeconds)
  } else {
    const previous = v053Rail.progress
    v053Rail.progress += deltaSeconds * 17.5 * v053Rail.direction / Math.max(v053Rail.routeLength, 1)
    if (v053Rail.progress >= 0.96) {
      v053Rail.progress = 0.96
      v053Rail.direction = -1
      v053Rail.dwell = 2.0
    } else if (v053Rail.progress <= 0.04) {
      v053Rail.progress = 0.04
      v053Rail.direction = 1
      v053Rail.dwell = 2.0
    } else {
      for (const station of v053Rail.stations) {
        if (v053CrossedStation(previous, v053Rail.progress, station, v053Rail.direction)) {
          v053Rail.progress = station
          v053Rail.dwell = 2.8
          break
        }
      }
    }
  }

  const vertices = []
  const routeLength = v053Rail.routeLength
  const spacing = 15.1
  for (let car = 0; car < 3; car += 1) {
    const progress = clamp(v053Rail.progress - car * spacing * v053Rail.direction / routeLength, 0.015, 0.985)
    const point = v053RailPointAt(v053Rail.route, progress)
    const dx = point.dx * v053Rail.direction
    const dz = point.dz * v053Rail.direction
    const rotation = Math.atan2(-dx, dz)
    v053PushRailCar(vertices, point.x, point.y + 1.18, point.z, rotation, car === 0, 1)
  }
  const data = new Float32Array(vertices)
  gl.bindBuffer(gl.ARRAY_BUFFER, v053RailBuffer)
  gl.bufferData(gl.ARRAY_BUFFER, data, gl.DYNAMIC_DRAW)
  v053RailVertexCount = data.length / 9
}

function resizeCanvas() {
  const pixelRatio = Math.min(window.devicePixelRatio || 1, 2)
  const width = Math.max(1, Math.floor(canvas.clientWidth * pixelRatio))
  const height = Math.max(1, Math.floor(canvas.clientHeight * pixelRatio))
  if (canvas.width !== width || canvas.height !== height) {
    canvas.width = width
    canvas.height = height
  }
  gl.viewport(0, 0, width, height)
}

function environmentForTime(time) {
  const daylight = smoothstep(6, 8, time) * (1 - smoothstep(18.2, 21.0, time))
  const sunset = Math.max(1 - Math.abs(time - 6.6) / 1.8, 1 - Math.abs(time - 18.8) / 2.0, 0)
  const night = 1 - daylight
  const skyDay = [0.61, 0.74, 0.82]
  const skyNight = [0.055, 0.085, 0.13]
  const skySunset = [0.78, 0.52, 0.39]
  let sky = mixColor(skyNight, skyDay, daylight)
  sky = mixColor(sky, skySunset, sunset * 0.38)
  const sunElevation = clamp(Math.sin(((time - 6) / 15) * Math.PI), 0.05, 1)
  const sunAzimuth = ((time - 6) / 17) * Math.PI * 1.35 - Math.PI * 0.25
  const lightDirection = normalize3([
    Math.cos(sunAzimuth) * (0.25 + sunElevation),
    0.18 + sunElevation * 1.10,
    Math.sin(sunAzimuth) * (0.25 + sunElevation),
  ])
  const sunColor = mixColor([0.95, 0.54, 0.30], [1.0, 0.98, 0.90], daylight)
  return {
    sky,
    lightDirection,
    sunColor,
    ambient: 0.20 + daylight * 0.33 + sunset * 0.04,
    nightFactor: clamp(night * 1.18, 0, 1),
  }
}

function niceScale(value) {
  const exponent = Math.floor(Math.log10(Math.max(value, 1)))
  const fraction = value / (10 ** exponent)
  const niceFraction = fraction < 1.5 ? 1 : fraction < 3.5 ? 2 : fraction < 7.5 ? 5 : 10
  return niceFraction * (10 ** exponent)
}

function updateScale() {
  const unitInMeters = window.ProceduralCityViewMode === 'world' ? 1000 : 1
  const rawMeters = camera.distance * 0.18 * unitInMeters
  const meters = niceScale(rawMeters)
  const width = clamp((meters / Math.max(rawMeters, 1)) * 90, 54, 118)
  ui.scaleLine.style.width = `${width}px`
  ui.scaleLabel.textContent = meters >= 1000 ? `${(meters / 1000).toFixed(meters % 1000 === 0 ? 0 : 1)} km` : `${meters} m`
  ui.compassNeedle.style.transform = `rotate(${-camera.yaw}rad)`
}

let previousFrameTime = performance.now()
function render(frameTime = performance.now()) {
  const deltaSeconds = Math.min(0.05, Math.max(0, (frameTime - previousFrameTime) / 1000))
  previousFrameTime = frameTime
  if (autoRotateEnabled && activePointers.size === 0) camera.yaw += deltaSeconds * 0.075

  resizeCanvas()
  const environment = environmentForTime(Number(ui.timeOfDay.value))
  gl.enable(gl.DEPTH_TEST)
  gl.disable(gl.CULL_FACE)
  gl.clearColor(environment.sky[0], environment.sky[1], environment.sky[2], 1)
  gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT)

  const cosPitch = Math.cos(camera.pitch)
  const eye = [
    camera.target[0] + Math.sin(camera.yaw) * cosPitch * camera.distance,
    camera.target[1] + Math.sin(camera.pitch) * camera.distance,
    camera.target[2] + Math.cos(camera.yaw) * cosPitch * camera.distance,
  ]
  const renderExtent = window.ProceduralCityRenderExtent || (currentCity ? currentCity.extent : 0)
  const far = Math.max(2500, renderExtent ? renderExtent * 4.2 : 3000)
  mat4Perspective(projection, Math.PI / 4, canvas.width / canvas.height, 0.2, far)
  mat4LookAt(view, eye, camera.target, [0, 1, 0])
  mat4Multiply(viewProjection, projection, view)

  gl.useProgram(program)
  gl.uniformMatrix4fv(locations.viewProjection, false, viewProjection)
  gl.uniform3fv(locations.lightDirection, environment.lightDirection)
  gl.uniform3fv(locations.sunColor, environment.sunColor)
  gl.uniform3fv(locations.fogColor, environment.sky)
  gl.uniform1f(locations.ambient, environment.ambient)
  gl.uniform1f(locations.fogNear, renderExtent ? renderExtent * 0.78 : 650)
  gl.uniform1f(locations.fogFar, renderExtent ? renderExtent * 2.55 : 1800)
  gl.uniform1f(locations.nightFactor, environment.nightFactor)
  if (locations.cameraPosition) gl.uniform3fv(locations.cameraPosition, eye)
  if (locations.time) gl.uniform1f(locations.time, frameTime * 0.001)
  if (locations.wetness) {
    const weather = document.querySelector('#weather-mode')?.value || 'auto'
    const wetness = weather === 'rain' ? 1 : weather === 'overcast' ? 0.38 : weather === 'auto' ? 0.12 : 0
    gl.uniform1f(locations.wetness, wetness)
  }

  v053BindGeometry(vertexBuffer)
  gl.drawArrays(gl.TRIANGLES, 0, vertexCount)

  if (window.ProceduralCityViewMode !== 'world') v053BuildTraffic(currentCity, deltaSeconds)
  if (window.ProceduralCityViewMode !== 'world' && v053TrafficVertexCount > 0) {
    v053BindGeometry(v053TrafficBuffer)
    gl.drawArrays(gl.TRIANGLES, 0, v053TrafficVertexCount)
  }

  if (window.ProceduralCityViewMode !== 'world') v053BuildRail(currentCity, deltaSeconds)
  if (window.ProceduralCityViewMode !== 'world' && v053RailVertexCount > 0) {
    v053BindGeometry(v053RailBuffer)
    gl.drawArrays(gl.TRIANGLES, 0, v053RailVertexCount)
  }

  updateScale()
  requestAnimationFrame(render)
}
