function dot3(a, b) {
  return a[0] * b[0] + a[1] * b[1] + a[2] * b[2]
}

function mat4LookAt(out, eye, target, up) {
  const zAxis = normalize3([eye[0] - target[0], eye[1] - target[1], eye[2] - target[2]])
  const xAxis = normalize3(cross3(up, zAxis))
  const yAxis = cross3(zAxis, xAxis)
  out[0] = xAxis[0]; out[1] = yAxis[0]; out[2] = zAxis[0]; out[3] = 0
  out[4] = xAxis[1]; out[5] = yAxis[1]; out[6] = zAxis[1]; out[7] = 0
  out[8] = xAxis[2]; out[9] = yAxis[2]; out[10] = zAxis[2]; out[11] = 0
  out[12] = -dot3(xAxis, eye); out[13] = -dot3(yAxis, eye); out[14] = -dot3(zAxis, eye); out[15] = 1
  return out
}

function mat4Multiply(out, a, b) {
  const result = new Float32Array(16)
  for (let column = 0; column < 4; column += 1) {
    for (let row = 0; row < 4; row += 1) {
      result[column * 4 + row] =
        a[row] * b[column * 4]
        + a[4 + row] * b[column * 4 + 1]
        + a[8 + row] * b[column * 4 + 2]
        + a[12 + row] * b[column * 4 + 3]
    }
  }
  out.set(result)
  return out
}

const projection = new Float32Array(16)
const view = new Float32Array(16)
const viewProjection = new Float32Array(16)

const transportBuffer = gl.createBuffer()
let transportAgents = []
let transportCity = null
const TRANSPORT_COLORS = [
  [0.78, 0.82, 0.84], [0.19, 0.42, 0.66], [0.72, 0.18, 0.14],
  [0.16, 0.56, 0.42], [0.86, 0.63, 0.16], [0.31, 0.32, 0.35],
]

function bindGeometryBuffer(buffer) {
  gl.bindBuffer(gl.ARRAY_BUFFER, buffer)
  const stride = 9 * Float32Array.BYTES_PER_ELEMENT
  gl.enableVertexAttribArray(locations.position)
  gl.vertexAttribPointer(locations.position, 3, gl.FLOAT, false, stride, 0)
  gl.enableVertexAttribArray(locations.normal)
  gl.vertexAttribPointer(locations.normal, 3, gl.FLOAT, false, stride, 3 * Float32Array.BYTES_PER_ELEMENT)
  gl.enableVertexAttribArray(locations.color)
  gl.vertexAttribPointer(locations.color, 3, gl.FLOAT, false, stride, 6 * Float32Array.BYTES_PER_ELEMENT)
}

function resetNativeTransport(city) {
  transportAgents = []
  transportCity = city
  if (!city) return
  const random = mulberry32(hashString(`${city.settings.seed}:native-transport`))
  const usableRoads = city.roads.filter((road) => road.class !== 'local')
  const count = Math.min(110, Math.max(42, Math.round(city.metrics.roadLengthKm * 4.8)))
  for (let index = 0; index < count; index += 1) {
    const road = usableRoads[Math.floor(random() * usableRoads.length)] || city.roads[index % city.roads.length]
    if (!road) continue
    const roll = random()
    transportAgents.push({
      kind: roll < 0.10 ? 'bus' : roll < 0.18 ? 'truck' : 'car',
      road,
      progress: random(),
      speed: 4.5 + random() * 8.5,
      direction: random() < 0.5 ? -1 : 1,
      lane: (random() < 0.5 ? -1 : 1) * (road.class === 'arterial' ? 5.8 : 2.5),
      color: TRANSPORT_COLORS[Math.floor(random() * TRANSPORT_COLORS.length)],
    })
  }
  const station = city.infrastructure.stations[Math.floor(city.infrastructure.stations.length / 2)]
  for (let index = 0; index < 3; index += 1) {
    transportAgents.push({
      kind: 'train',
      progress: index / 3,
      speed: 15 + index * 1.6,
      direction: index % 2 ? -1 : 1,
      x: city.centers.cbd.x,
      y: station ? station.y + 0.62 : 2.2,
      extent: city.extent * 0.92,
      color: index % 2 ? [0.18, 0.63, 0.79] : [0.82, 0.36, 0.18],
    })
  }
  document.body.classList.add('native-webgl-transport')
  window.dispatchEvent(new CustomEvent('procedural-city:native-transport-ready', {
    detail: { agents: transportAgents.length, mode: 'webgl' },
  }))
}

function pushWheel(vertices, x, y, z, rotation, along, across) {
  const [ox, oz] = rotation === 0 ? [across, along] : [along, across]
  pushBox(vertices, x + ox, y, z + oz, 0.48, 0.55, 0.24, [0.035, 0.04, 0.045], rotation)
}

function pushRoadVehicle(vertices, agent, x, y, z, rotation) {
  const isBus = agent.kind === 'bus'
  const isTruck = agent.kind === 'truck'
  const length = isBus ? 10.6 : isTruck ? 8.3 : 4.7
  const width = isBus ? 2.55 : isTruck ? 2.45 : 1.92
  const height = isBus ? 2.85 : isTruck ? 2.75 : 1.35
  const bodyY = y + 0.28
  pushBox(vertices, x, bodyY, z, width, height * 0.58, length, agent.color, rotation)
  if (isTruck) {
    const cabOffset = length * 0.31
    const dx = Math.sin(rotation) * cabOffset
    const dz = Math.cos(rotation) * cabOffset
    pushBox(vertices, x + dx, bodyY + height * 0.45, z + dz, width * 0.94, height * 0.48, length * 0.28, tintColor(agent.color, 0.08), rotation)
    pushBox(vertices, x - dx * 0.32, bodyY + height * 0.58, z - dz * 0.32, width * 0.92, height * 0.48, length * 0.58, tintColor(agent.color, -0.10), rotation)
  } else {
    pushBox(vertices, x, bodyY + height * 0.52, z, width * 0.82, height * 0.42, length * (isBus ? 0.88 : 0.53), [0.18, 0.29, 0.35], rotation)
  }
  const wheelAlong = length * 0.31
  const wheelAcross = width * 0.52
  pushWheel(vertices, x, bodyY + 0.03, z, rotation, wheelAlong, wheelAcross)
  pushWheel(vertices, x, bodyY + 0.03, z, rotation, wheelAlong, -wheelAcross)
  pushWheel(vertices, x, bodyY + 0.03, z, rotation, -wheelAlong, wheelAcross)
  pushWheel(vertices, x, bodyY + 0.03, z, rotation, -wheelAlong, -wheelAcross)
  const lampOffset = length * 0.51
  const lx = Math.sin(rotation) * lampOffset
  const lz = Math.cos(rotation) * lampOffset
  pushBox(vertices, x + lx, bodyY + height * 0.25, z + lz, width * 0.58, 0.13, 0.08, [1.0, 0.85, 0.47], rotation)
}

function pushTrain(vertices, agent, z, direction) {
  const rotation = 0
  for (let car = -1; car <= 1; car += 1) {
    const carZ = z + car * 14.5 * direction
    pushBox(vertices, agent.x, agent.y, carZ, 3.25, 3.15, 13.3, agent.color, rotation)
    pushBox(vertices, agent.x, agent.y + 1.22, carZ, 3.31, 1.18, 10.5, [0.16, 0.30, 0.38], rotation)
    pushBox(vertices, agent.x, agent.y + 3.14, carZ, 2.2, 0.18, 7.8, tintColor(agent.color, 0.16), rotation)
  }
}

function buildTransportVertices(deltaSeconds) {
  if (currentCity !== transportCity) resetNativeTransport(currentCity)
  const vertices = []
  for (const agent of transportAgents) {
    const routeLength = agent.kind === 'train' ? agent.extent : Math.max(agent.road.length, 1)
    agent.progress = (agent.progress + deltaSeconds * agent.speed * agent.direction / routeLength + 10) % 1
    if (agent.kind === 'train') {
      const z = -agent.extent / 2 + agent.progress * agent.extent
      pushTrain(vertices, agent, z, agent.direction)
      continue
    }
    const offset = (agent.progress - 0.5) * agent.road.length
    let x = agent.road.x
    let z = agent.road.z
    let rotation = 0
    if (agent.road.orientation === 'vertical') {
      x += agent.lane
      z += offset
      rotation = agent.direction > 0 ? 0 : Math.PI
    } else {
      x += offset
      z += agent.lane
      rotation = agent.direction > 0 ? Math.PI / 2 : -Math.PI / 2
    }
    pushRoadVehicle(vertices, agent, x, agent.road.y + 0.22, z, rotation)
  }
  return new Float32Array(vertices)
}

function resizeCanvas() {
  const pixelRatio = Math.min(window.devicePixelRatio || 1, 2)
  const width = Math.max(1, Math.floor(canvas.clientWidth * pixelRatio))
  const height = Math.max(1, Math.floor(canvas.clientHeight * pixelRatio))
  if (canvas.width !== width || canvas.height !== height) {
    canvas.width = width
    canvas.height = height
  }
  gl.viewport(0, 0, width, height)
}

function environmentForTime(time) {
  const daylight = smoothstep(6, 8, time) * (1 - smoothstep(18.2, 21.0, time))
  const sunset = Math.max(
    1 - Math.abs(time - 6.6) / 1.8,
    1 - Math.abs(time - 18.8) / 2.0,
    0,
  )
  const night = 1 - daylight
  const skyDay = [0.61, 0.74, 0.82]
  const skyNight = [0.055, 0.085, 0.13]
  const skySunset = [0.78, 0.52, 0.39]
  let sky = mixColor(skyNight, skyDay, daylight)
  sky = mixColor(sky, skySunset, sunset * 0.38)
  const sunElevation = clamp(Math.sin(((time - 6) / 15) * Math.PI), 0.05, 1)
  const sunAzimuth = ((time - 6) / 17) * Math.PI * 1.35 - Math.PI * 0.25
  const lightDirection = normalize3([
    Math.cos(sunAzimuth) * (0.25 + sunElevation),
    0.18 + sunElevation * 1.10,
    Math.sin(sunAzimuth) * (0.25 + sunElevation),
  ])
  const sunColor = mixColor([0.95, 0.54, 0.30], [1.0, 0.98, 0.90], daylight)
  return {
    sky,
    lightDirection,
    sunColor,
    ambient: 0.20 + daylight * 0.33 + sunset * 0.04,
    nightFactor: clamp(night * 1.18, 0, 1),
  }
}

function niceScale(value) {
  const exponent = Math.floor(Math.log10(Math.max(value, 1)))
  const fraction = value / (10 ** exponent)
  const niceFraction = fraction < 1.5 ? 1 : fraction < 3.5 ? 2 : fraction < 7.5 ? 5 : 10
  return niceFraction * (10 ** exponent)
}

function updateScale() {
  const meters = niceScale(camera.distance * 0.18)
  const width = clamp((meters / Math.max(camera.distance * 0.18, 1)) * 90, 54, 118)
  ui.scaleLine.style.width = `${width}px`
  ui.scaleLabel.textContent = meters >= 1000 ? `${(meters / 1000).toFixed(meters % 1000 === 0 ? 0 : 1)} km` : `${meters} m`
  ui.compassNeedle.style.transform = `rotate(${-camera.yaw}rad)`
}

let previousFrameTime = performance.now()
function render(frameTime = performance.now()) {
  const deltaSeconds = Math.min(0.05, (frameTime - previousFrameTime) / 1000)
  previousFrameTime = frameTime
  if (autoRotateEnabled && activePointers.size === 0) camera.yaw += deltaSeconds * 0.075

  resizeCanvas()
  const environment = environmentForTime(Number(ui.timeOfDay.value))
  gl.enable(gl.DEPTH_TEST)
  gl.disable(gl.CULL_FACE)
  gl.clearColor(environment.sky[0], environment.sky[1], environment.sky[2], 1)
  gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT)

  const cosPitch = Math.cos(camera.pitch)
  const eye = [
    camera.target[0] + Math.sin(camera.yaw) * cosPitch * camera.distance,
    camera.target[1] + Math.sin(camera.pitch) * camera.distance,
    camera.target[2] + Math.cos(camera.yaw) * cosPitch * camera.distance,
  ]
  const far = Math.max(2500, currentCity ? currentCity.extent * 4.2 : 3000)
  mat4Perspective(projection, Math.PI / 4, canvas.width / canvas.height, 0.2, far)
  mat4LookAt(view, eye, camera.target, [0, 1, 0])
  mat4Multiply(viewProjection, projection, view)

  gl.useProgram(program)
  gl.uniformMatrix4fv(locations.viewProjection, false, viewProjection)
  gl.uniform3fv(locations.lightDirection, environment.lightDirection)
  gl.uniform3fv(locations.sunColor, environment.sunColor)
  gl.uniform3fv(locations.fogColor, environment.sky)
  gl.uniform1f(locations.ambient, environment.ambient)
  gl.uniform1f(locations.fogNear, currentCity ? currentCity.extent * 0.78 : 650)
  gl.uniform1f(locations.fogFar, currentCity ? currentCity.extent * 2.55 : 1800)
  gl.uniform1f(locations.nightFactor, environment.nightFactor)

  bindGeometryBuffer(vertexBuffer)
  gl.drawArrays(gl.TRIANGLES, 0, vertexCount)

  const transportVertices = buildTransportVertices(deltaSeconds)
  if (transportVertices.length) {
    gl.bindBuffer(gl.ARRAY_BUFFER, transportBuffer)
    gl.bufferData(gl.ARRAY_BUFFER, transportVertices, gl.DYNAMIC_DRAW)
    bindGeometryBuffer(transportBuffer)
    gl.drawArrays(gl.TRIANGLES, 0, transportVertices.length / 9)
  }

  updateScale()
  requestAnimationFrame(render)
}
