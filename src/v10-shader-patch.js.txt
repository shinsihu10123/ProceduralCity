// Procedural City v1.1 — material-aware physical lighting and procedural facade shader.
// Loaded before the legacy engine so its WebGL shader source can be upgraded
// without changing the renderer's existing uniforms or draw contract.
(function () {
  const originalShaderSource = WebGL2RenderingContext.prototype.shaderSource

  const enhancedFragmentShader = `#version 300 es
precision highp float;
in vec3 vWorldPosition;
in vec3 vNormal;
in vec3 vColor;
in float vFogDepth;
uniform vec3 uLightDirection;
uniform vec3 uSunColor;
uniform vec3 uFogColor;
uniform float uAmbient;
uniform float uFogNear;
uniform float uFogFar;
uniform float uNightFactor;
uniform vec3 uCameraPosition;
uniform float uTime;
uniform float uWetness;
out vec4 outColor;

float saturate(float value) { return clamp(value, 0.0, 1.0); }
float hash21(vec2 point) { return fract(sin(dot(point, vec2(127.1, 311.7))) * 43758.5453123); }
vec3 acesTone(vec3 value) {
  return clamp((value * (2.51 * value + 0.03)) / (value * (2.43 * value + 0.59) + 0.14), 0.0, 1.0);
}

void main() {
  vec3 normal = normalize(vNormal);
  vec3 lightDir = normalize(uLightDirection);
  vec3 viewDir = normalize(uCameraPosition - vWorldPosition);
  vec3 halfDir = normalize(lightDir + viewDir);

  float ndl = max(dot(normal, lightDir), 0.0);
  float ndh = max(dot(normal, halfDir), 0.0);
  float upFacing = saturate(normal.y * 0.5 + 0.5);
  float horizontalSurface = smoothstep(0.72, 0.96, normal.y);
  float verticalSurface = 1.0 - smoothstep(0.24, 0.76, abs(normal.y));

  float luminance = dot(vColor, vec3(0.299, 0.587, 0.114));
  float blueBias = vColor.b - (vColor.r + vColor.g) * 0.5;
  float greenBias = vColor.g - max(vColor.r, vColor.b);
  float chroma = max(vColor.r, max(vColor.g, vColor.b)) - min(vColor.r, min(vColor.g, vColor.b));

  float blueMaterial = saturate(smoothstep(0.012, 0.13, blueBias) * smoothstep(0.16, 0.72, luminance));
  float glassMask = blueMaterial * verticalSurface;
  float waterMask = blueMaterial * horizontalSurface;
  float vegetationMask = saturate(smoothstep(0.025, 0.18, greenBias) * smoothstep(0.12, 0.62, luminance));
  float asphaltMask = saturate((1.0 - smoothstep(0.13, 0.27, luminance)) * (1.0 - smoothstep(0.08, 0.22, chroma)) * upFacing);
  float metalMask = saturate((1.0 - blueMaterial) * smoothstep(0.24, 0.58, luminance) * (1.0 - smoothstep(0.05, 0.2, chroma)));
  float emissiveWindow = smoothstep(0.70, 0.92, luminance) * smoothstep(0.12, 0.52, vColor.r - vColor.b);
  float concreteMask = verticalSurface * smoothstep(0.31, 0.55, luminance) * (1.0 - glassMask) * (1.0 - vegetationMask);

  float facadeAxis = mix(vWorldPosition.x, vWorldPosition.z, step(abs(normal.z), abs(normal.x)));
  float facadeSeed = hash21(floor(vWorldPosition.xz * 0.075));
  float bayCell = fract((facadeAxis + facadeSeed * 1.9) / 3.05);
  float floorCell = fract((vWorldPosition.y + facadeSeed * 0.45) / 3.20);
  float bayOpening = smoothstep(0.11, 0.18, bayCell) * (1.0 - smoothstep(0.82, 0.89, bayCell));
  float floorOpening = smoothstep(0.20, 0.27, floorCell) * (1.0 - smoothstep(0.75, 0.82, floorCell));
  float proceduralWindow = concreteMask * bayOpening * floorOpening;
  float interiorVariation = 0.50 + hash21(floor(vWorldPosition.xz * 0.33) + floor(vWorldPosition.yy * 0.31)) * 0.50;

  float roughness = mix(0.76, 0.22, glassMask);
  roughness = mix(roughness, 0.34, metalMask);
  roughness = mix(roughness, mix(0.62, 0.26, uWetness), asphaltMask);
  roughness = mix(roughness, 0.88, vegetationMask);
  roughness = mix(roughness, 0.18, waterMask);

  float specularPower = mix(10.0, 112.0, 1.0 - roughness);
  float specular = pow(ndh, specularPower);
  float fresnel = pow(1.0 - max(dot(normal, viewDir), 0.0), 5.0);
  float glassReflection = glassMask * (0.16 + fresnel * 0.72);
  float asphaltReflection = asphaltMask * pow(ndh, mix(18.0, 54.0, uWetness)) * (0.08 + uWetness * 0.44 + uNightFactor * 0.18);
  float metalReflection = metalMask * specular * 0.34;
  float waterRipple = sin(vWorldPosition.x * 0.17 + uTime * 0.48) * sin(vWorldPosition.z * 0.13 - uTime * 0.36) * 0.5 + 0.5;
  float waterReflection = waterMask * (0.20 + fresnel * 0.62 + waterRipple * 0.08);

  float hemisphere = mix(0.42, 1.0, upFacing);
  float terrainHeightShade = clamp(0.90 + vWorldPosition.y * 0.00078, 0.78, 1.06);
  vec3 ambientLight = vec3(uAmbient) * mix(vec3(0.78, 0.86, 0.94), vec3(1.0), upFacing);
  vec3 directLight = uSunColor * ndl * 0.79;
  float microVariation = 0.965 + hash21(floor(vWorldPosition.xz * 0.42)) * 0.07;
  vec3 baseColor = mix(vColor, vec3(0.12, 0.23, 0.31), proceduralWindow * 0.78);
  vec3 diffuseColor = baseColor * (ambientLight + directLight) * hemisphere * terrainHeightShade * microVariation;

  vec3 skyReflection = mix(uFogColor, vec3(0.68, 0.79, 0.91), 0.42 + upFacing * 0.3);
  vec3 reflected = skyReflection * glassReflection
    + uSunColor * metalReflection
    + mix(uFogColor, uSunColor, 0.35) * asphaltReflection
    + mix(vec3(0.16, 0.32, 0.44), skyReflection, 0.66) * waterReflection;

  float vegetationVariation = 0.93 + 0.07 * sin(vWorldPosition.x * 0.17 + vWorldPosition.z * 0.13);
  diffuseColor *= mix(1.0, vegetationVariation, vegetationMask);

  float proceduralEmission = proceduralWindow * uNightFactor * step(0.36, interiorVariation) * interiorVariation;
  vec3 warmEmission = vec3(1.0, 0.63, 0.25) * (emissiveWindow * 1.42 + proceduralEmission * 0.92);
  vec3 coolGlass = vec3(0.18, 0.34, 0.52) * glassMask * (0.08 + uNightFactor * 0.12);
  vec3 lit = diffuseColor + reflected + warmEmission + coolGlass;

  float distanceFog = smoothstep(uFogNear, uFogFar, vFogDepth);
  float altitudeHaze = saturate((24.0 - vWorldPosition.y) / 150.0) * distanceFog * 0.16;
  float aerialPerspective = saturate(distanceFog * 0.84 + altitudeHaze);
  vec3 fogged = mix(lit, uFogColor, aerialPerspective);

  vec3 mapped = acesTone(fogged * 1.08);
  mapped = pow(max(mapped, vec3(0.0)), vec3(1.0 / 2.2));
  outColor = vec4(mapped, 1.0);
}`

  WebGL2RenderingContext.prototype.shaderSource = function patchedShaderSource(shader, source) {
    if (typeof source === 'string' && source.includes('uniform float uNightFactor;') && source.includes('warmWindow')) {
      return originalShaderSource.call(this, shader, enhancedFragmentShader)
    }
    return originalShaderSource.call(this, shader, source)
  }

  window.ProceduralCityV10ShaderPatch = {
    version: '1.1.0-material-facade-lighting',
    active: true,
  }
})()
