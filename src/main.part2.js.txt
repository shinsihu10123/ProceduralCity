  ]
}

function pushVertex(target, position, normal, color) {
  target.push(...position, ...normal, ...color)
}

function pushTriangle(target, a, b, c, normal, color) {
  pushVertex(target, a, normal, color)
  pushVertex(target, b, normal, color)
  pushVertex(target, c, normal, color)
}

function pushQuad(target, a, b, c, d, normal, color) {
  pushTriangle(target, a, b, c, normal, color)
  pushTriangle(target, a, c, d, normal, color)
}

function pushBox(target, x, y, z, width, height, depth, color, rotation = 0, roofColor = null) {
  const halfWidth = width / 2
  const halfDepth = depth / 2
  const cos = Math.cos(rotation)
  const sin = Math.sin(rotation)
  const transform = (localX, localY, localZ) => [
    x + localX * cos - localZ * sin,
    y + localY,
    z + localX * sin + localZ * cos,
  ]
  const rotateNormal = (nx, ny, nz) => [nx * cos - nz * sin, ny, nx * sin + nz * cos]

  const p000 = transform(-halfWidth, 0, -halfDepth)
  const p100 = transform(halfWidth, 0, -halfDepth)
  const p110 = transform(halfWidth, height, -halfDepth)
  const p010 = transform(-halfWidth, height, -halfDepth)
  const p001 = transform(-halfWidth, 0, halfDepth)
  const p101 = transform(halfWidth, 0, halfDepth)
  const p111 = transform(halfWidth, height, halfDepth)
  const p011 = transform(-halfWidth, height, halfDepth)

  pushQuad(target, p000, p100, p110, p010, rotateNormal(0, 0, -1), color)
  pushQuad(target, p101, p001, p011, p111, rotateNormal(0, 0, 1), color)
  pushQuad(target, p001, p000, p010, p011, rotateNormal(-1, 0, 0), color)
  pushQuad(target, p100, p101, p111, p110, rotateNormal(1, 0, 0), color)
  pushQuad(target, p010, p110, p111, p011, [0, 1, 0], roofColor || color)
}

function pushTerrain(target, settings, extent, seedOffset) {
  const size = extent * 1.35
  const segments = Math.min(72, settings.gridSize * 2)
  const step = size / segments
  const start = -size / 2

  for (let xIndex = 0; xIndex < segments; xIndex += 1) {
    for (let zIndex = 0; zIndex < segments; zIndex += 1) {
      const x0 = start + xIndex * step
      const z0 = start + zIndex * step
      const x1 = x0 + step
      const z1 = z0 + step
      const y00 = terrainHeight(x0, z0, seedOffset, settings.roughness, extent)
      const y10 = terrainHeight(x1, z0, seedOffset, settings.roughness, extent)
      const y11 = terrainHeight(x1, z1, seedOffset, settings.roughness, extent)
      const y01 = terrainHeight(x0, z1, seedOffset, settings.roughness, extent)
      const average = (y00 + y10 + y11 + y01) / 4
      const colorFactor = Math.min(1, Math.max(0, average / 5 + 0.5))
      const color = mixColor(COLORS.terrainLow, COLORS.terrainHigh, colorFactor)
      const normal = calculateNormal([x0, y00, z0], [x1, y10, z0], [x1, y11, z1])
      const normal2 = calculateNormal([x0, y00, z0], [x1, y11, z1], [x0, y01, z1])
      pushTriangle(target, [x0, y00, z0], [x1, y10, z0], [x1, y11, z1], normal, color)
      pushTriangle(target, [x0, y00, z0], [x1, y11, z1], [x0, y01, z1], normal2, color)
    }
  }
}

function calculateNormal(a, b, c) {
  const ux = b[0] - a[0]
  const uy = b[1] - a[1]
  const uz = b[2] - a[2]
  const vx = c[0] - a[0]
  const vy = c[1] - a[1]
  const vz = c[2] - a[2]
  let nx = uy * vz - uz * vy
  let ny = uz * vx - ux * vz
  let nz = ux * vy - uy * vx
  const length = Math.hypot(nx, ny, nz) || 1
  nx /= length
  ny /= length
  nz /= length
  if (ny < 0) {
    nx *= -1
    ny *= -1
    nz *= -1
  }
  return [nx, ny, nz]
}

function chooseZone(distanceRatio, random) {
  const roll = random()
  if (distanceRatio < 0.22) return roll < 0.78 ? 'commercial' : 'civic'
  if (distanceRatio > 0.72) return roll < 0.66 ? 'residential' : 'industrial'
  if (roll < 0.58) return 'residential'
  if (roll < 0.78) return 'commercial'
  if (roll < 0.93) return 'industrial'
  return 'civic'
}

function buildingHeight(zone, centerFactor, random) {
  const base = {
    residential: range(random, 3.2, 11),
    commercial: range(random, 7, 23),
    industrial: range(random, 2.8, 7),
    civic: range(random, 4, 12),
  }[zone]
  return Math.max(2.5, base * (0.55 + centerFactor * 1.15))
}
