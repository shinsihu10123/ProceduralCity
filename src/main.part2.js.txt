      pushTriangle(target, [x0, y00, z0], [x1, y11, z1], [x0, y01, z1], normal2, color)
    }
  }
}

function calculateNormal(a, b, c) {
  const ux = b[0] - a[0]
  const uy = b[1] - a[1]
  const uz = b[2] - a[2]
  const vx = c[0] - a[0]
  const vy = c[1] - a[1]
  const vz = c[2] - a[2]
  let nx = uy * vz - uz * vy
  let ny = uz * vx - ux * vz
  let nz = ux * vy - uy * vx
  const length = Math.hypot(nx, ny, nz) || 1
  nx /= length
  ny /= length
  nz /= length
  if (ny < 0) {
    nx *= -1
    ny *= -1
    nz *= -1
  }
  return [nx, ny, nz]
}

const ROAD_CLASSES = {
  arterial: { width: 9.4, sidewalk: 2.8, color: COLORS.roadArterial },
  collector: { width: 7.0, sidewalk: 2.4, color: COLORS.roadCollector },
  local: { width: 5.2, sidewalk: 2.0, color: COLORS.roadLocal },
}

function buildRoadLines(gridSize, spacing) {
  const center = Math.floor(gridSize / 2)
  const lines = new Map()
  for (let index = center; index >= 0; index -= spacing) {
    const offset = center - index
    lines.set(index, offset % (spacing * 3) === 0 ? 'arterial' : (offset % (spacing * 2) === 0 ? 'collector' : 'local'))
  }
  for (let index = center + spacing; index < gridSize; index += spacing) {
    const offset = index - center
    lines.set(index, offset % (spacing * 3) === 0 ? 'arterial' : (offset % (spacing * 2) === 0 ? 'collector' : 'local'))
  }
  return lines
}

function nearestLine(index, lines, acceptedClasses = null) {
  let best = { distance: Infinity, index: null, roadClass: null }
  for (const [lineIndex, roadClass] of lines.entries()) {
    if (acceptedClasses && !acceptedClasses.includes(roadClass)) continue
    const distance = Math.abs(index - lineIndex)
    if (distance < best.distance) best = { distance, index: lineIndex, roadClass }
  }
  return best
}

function createDistrictPlan(random, extent) {
  const cbd = [range(random, -extent * 0.08, extent * 0.08), range(random, -extent * 0.08, extent * 0.08)]
  const secondaryCenters = []
  for (let index = 0; index < 2; index += 1) {
    const angle = range(random, 0, Math.PI * 2)
    const radius = range(random, extent * 0.20, extent * 0.34)
    secondaryCenters.push([Math.cos(angle) * radius, Math.sin(angle) * radius])
  }
  const industrialAngle = range(random, 0, Math.PI * 2)
  const industrial = [
    Math.cos(industrialAngle) * extent * 0.43,
    Math.sin(industrialAngle) * extent * 0.43,
  ]
  const parkCenters = []
  for (let index = 0; index < 3; index += 1) {
    const angle = range(random, 0, Math.PI * 2)
    const radius = range(random, extent * 0.12, extent * 0.37)
    parkCenters.push({
      x: Math.cos(angle) * radius,
      z: Math.sin(angle) * radius,
      radius: range(random, extent * 0.055, extent * 0.095),
    })
  }
  const civicCenters = [
    [
      cbd[0] + range(random, -extent * 0.08, extent * 0.08),
      cbd[1] + range(random, -extent * 0.08, extent * 0.08),
    ],
    [
      secondaryCenters[0][0] + range(random, -extent * 0.045, extent * 0.045),
      secondaryCenters[0][1] + range(random, -extent * 0.045, extent * 0.045),
    ],
    [
      secondaryCenters[1][0] + range(random, -extent * 0.045, extent * 0.045),
      secondaryCenters[1][1] + range(random, -extent * 0.045, extent * 0.045),
    ],
  ]
  return { cbd, secondaryCenters, industrial, parkCenters, civicCenters }
}

function pointInfluence(x, z, point, radius) {
  return 1 - clamp(Math.hypot(x - point[0], z - point[1]) / radius, 0, 1)
}

function parkInfluenceAt(x, z, plan) {
  let influence = 0
  for (const park of plan.parkCenters) {
    influence = Math.max(influence, 1 - clamp(Math.hypot(x - park.x, z - park.z) / park.radius, 0, 1))
  }
  return influence
}

function chooseZone(context, plan, random) {
  const cbdInfluence = pointInfluence(context.x, context.z, plan.cbd, context.extent * 0.30)
  let secondaryInfluence = 0
  for (const center of plan.secondaryCenters) {
    secondaryInfluence = Math.max(secondaryInfluence, pointInfluence(context.x, context.z, center, context.extent * 0.20))
  }
  const industrialInfluence = pointInfluence(context.x, context.z, plan.industrial, context.extent * 0.23)
  let civicInfluence = 0
  for (const center of plan.civicCenters) {
    civicInfluence = Math.max(civicInfluence, pointInfluence(context.x, context.z, center, context.extent * 0.10))
  }
  const edgeFactor = clamp(Math.hypot(context.x, context.z) / (context.extent * 0.60), 0, 1)

  if (civicInfluence > 0.30 && random() < 0.46) return 'civic'

  const scores = {
    residential: 0.62 + secondaryInfluence * 0.44 + context.parkAmenity * 0.34 - cbdInfluence * 0.38 - industrialInfluence * 0.95,
    commercial: 0.13 + cbdInfluence * 1.32 + secondaryInfluence * 0.58 + context.arterialAccess * 0.42 - industrialInfluence * 0.28,
    industrial: 0.02 + industrialInfluence * 1.42 + edgeFactor * 0.40 + context.arterialAccess * 0.18 - cbdInfluence * 0.82,
    civic: 0.02 + civicInfluence * 0.92 + context.parkAmenity * 0.16,
  }

  let chosen = 'residential'
  let bestScore = -Infinity
  for (const [zone, score] of Object.entries(scores)) {
    const jitter = range(random, -0.10, 0.10)
    if (score + jitter > bestScore) {
      chosen = zone
      bestScore = score + jitter
    }
  }
  return chosen
}

function createBuildingSpec(zone, context, random) {
  const densityBias = (context.settings.density - 0.35) / 0.60
  const intensity = clamp(
    0.10 + context.cbdInfluence * 0.78 + context.secondaryInfluence * 0.30
    + context.arterialAccess * 0.20 + densityBias * 0.28 - context.slope / 42,
    0,
    1,
  )
  const streetRotation = context.nearestRoadOrientation === 'horizontal' ? Math.PI / 2 : 0
  const colorVariation = range(random, -0.08, 0.10)
  const baseColor = mixColor(COLORS[zone], colorVariation >= 0 ? [0.95, 0.95, 0.92] : [0.18, 0.20, 0.21], Math.abs(colorVariation))

  let type
  let floors
  let floorHeight
  let width
  let depth

  if (zone === 'residential') {
    if (intensity < 0.36) {
      type = 'detached'
      floors = Math.round(range(random, 2, 4))
      floorHeight = 3.0
      width = range(random, 8.0, 10.8)
      depth = range(random, 8.5, 11.5)
    } else if (intensity < 0.70) {
      type = 'apartment-slab'
      floors = Math.round(range(random, 5, 12) * (0.78 + intensity * 0.36))
      floorHeight = 3.05
      width = range(random, 12.0, 15.2)
      depth = range(random, 6.8, 8.8)
    } else {
      type = 'residential-tower'
      floors = Math.round(range(random, 12, 25) * (0.76 + intensity * 0.42))
      floorHeight = 3.15
      width = range(random, 9.5, 12.2)
      depth = range(random, 9.5, 12.2)
    }
  } else if (zone === 'commercial') {
    if (intensity > 0.62) {
      type = 'office-tower'
      floors = Math.round(range(random, 12, 31) * (0.72 + intensity * 0.50))
      floorHeight = 3.75
      width = range(random, 10.0, 13.0)
      depth = range(random, 10.0, 13.0)
    } else {
      type = 'mixed-use'
      floors = Math.round(range(random, 4, 10) * (0.82 + intensity * 0.28))
      floorHeight = 3.45
      width = range(random, 12.5, 15.0)
      depth = range(random, 10.0, 13.8)
    }
  } else if (zone === 'industrial') {
    type = 'industrial-hall'
    floors = Math.round(range(random, 1, 3))
    floorHeight = 5.0
    width = range(random, 13.5, 16.0)
    depth = range(random, 12.5, 16.0)
  } else {
    type = 'civic-complex'
    floors = Math.round(range(random, 2, 6))
    floorHeight = 3.7
    width = range(random, 12.0, 15.5)
    depth = range(random, 10.0, 14.5)
  }

  const height = floors * floorHeight
  const efficiency = zone === 'industrial' ? 0.82 : 0.76
  return {
    type,
    floors,
    floorHeight,
    width,
    depth,
    height,
    rotation: streetRotation,
    color: baseColor,
    floorArea: width * depth * floors * efficiency,
    intensity,
  }
}

function pushTree(target, x, y, z, random, scale = 1) {
  const trunkHeight = range(random, 2.4, 3.4) * scale
  pushBox(target, x, y, z, 0.48 * scale, trunkHeight, 0.48 * scale, COLORS.treeTrunk)

}
