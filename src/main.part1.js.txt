const VERTEX_SHADER = `#version 300 es
precision highp float;
in vec3 aPosition;
in vec3 aNormal;
in vec3 aColor;
uniform mat4 uViewProjection;
out vec3 vWorldPosition;
out vec3 vNormal;
out vec3 vColor;
out float vFogDepth;
void main() {
  vec4 worldPosition = vec4(aPosition, 1.0);
  gl_Position = uViewProjection * worldPosition;
  vWorldPosition = aPosition;
  vNormal = aNormal;
  vColor = aColor;
  vFogDepth = gl_Position.w;
}`

const FRAGMENT_SHADER = `#version 300 es
precision highp float;
in vec3 vWorldPosition;
in vec3 vNormal;
in vec3 vColor;
in float vFogDepth;
uniform vec3 uLightDirection;
uniform vec3 uSunColor;
uniform vec3 uFogColor;
uniform float uAmbient;
uniform float uFogNear;
uniform float uFogFar;
uniform float uNightFactor;
out vec4 outColor;
void main() {
  vec3 normal = normalize(vNormal);
  float diffuse = max(dot(normal, normalize(uLightDirection)), 0.0);
  float skyLight = 0.45 + max(normal.y, 0.0) * 0.55;
  float groundShade = clamp(0.82 + vWorldPosition.y * 0.0012, 0.78, 1.04);
  vec3 lit = vColor * (vec3(uAmbient) + diffuse * 0.72 * uSunColor) * skyLight * groundShade;
  float luminance = dot(vColor, vec3(0.299, 0.587, 0.114));
  float warmWindow = smoothstep(0.73, 0.93, luminance) * smoothstep(0.18, 0.55, vColor.r - vColor.b);
  vec3 emissive = vec3(1.0, 0.71, 0.34) * warmWindow * uNightFactor * 0.85;
  float fogFactor = smoothstep(uFogNear, uFogFar, vFogDepth);
  vec3 finalColor = mix(lit + emissive, uFogColor, fogFactor);
  outColor = vec4(finalColor, 1.0);
}`

function compileShader(type, source) {
  const shader = gl.createShader(type)
  gl.shaderSource(shader, source)
  gl.compileShader(shader)
  if (!gl.getShaderParameter(shader, gl.COMPILE_STATUS)) {
    throw new Error(gl.getShaderInfoLog(shader) || 'Shader compile error')
  }
  return shader
}

function createProgram() {
  const program = gl.createProgram()
  gl.attachShader(program, compileShader(gl.VERTEX_SHADER, VERTEX_SHADER))
  gl.attachShader(program, compileShader(gl.FRAGMENT_SHADER, FRAGMENT_SHADER))
  gl.linkProgram(program)
  if (!gl.getProgramParameter(program, gl.LINK_STATUS)) {
    throw new Error(gl.getProgramInfoLog(program) || 'Program link error')
  }
  return program
}

function hashString(value) {
  let hash = 2166136261
  for (let index = 0; index < value.length; index += 1) {
    hash ^= value.charCodeAt(index)
    hash = Math.imul(hash, 16777619)
  }
  return hash >>> 0
}

function mulberry32(seed) {
  let state = seed >>> 0
  return () => {
    state += 0x6d2b79f5
    let value = state
    value = Math.imul(value ^ (value >>> 15), value | 1)
    value ^= value + Math.imul(value ^ (value >>> 7), value | 61)
    return ((value ^ (value >>> 14)) >>> 0) / 4294967296
  }
}

function clamp(value, min, max) {
  return Math.min(max, Math.max(min, value))
}

function lerp(a, b, factor) {
  return a + (b - a) * factor
}

function smoothstep(edge0, edge1, value) {
  const t = clamp((value - edge0) / (edge1 - edge0), 0, 1)
  return t * t * (3 - 2 * t)
}

function range(random, min, max) {
  return min + (max - min) * random()
}

function pick(random, values) {
  return values[Math.min(values.length - 1, Math.floor(random() * values.length))]
}

function hash2(x, z, seed) {
  let value = Math.imul(x, 374761393) + Math.imul(z, 668265263) + Math.imul(seed, 1442695041)
  value = (value ^ (value >>> 13)) * 1274126177
  return ((value ^ (value >>> 16)) >>> 0) / 4294967295
}

function valueNoise(x, z, seed) {
  const x0 = Math.floor(x)
  const z0 = Math.floor(z)
  const fx = x - x0
  const fz = z - z0
  const ux = fx * fx * (3 - 2 * fx)
  const uz = fz * fz * (3 - 2 * fz)
  const a = hash2(x0, z0, seed)
  const b = hash2(x0 + 1, z0, seed)
  const c = hash2(x0, z0 + 1, seed)
  const d = hash2(x0 + 1, z0 + 1, seed)
  return lerp(lerp(a, b, ux), lerp(c, d, ux), uz) * 2 - 1
}

function fbm(x, z, seed, octaves = 4) {
  let value = 0
  let amplitude = 0.55
  let frequency = 1
  let normalizer = 0
  for (let octave = 0; octave < octaves; octave += 1) {
    value += valueNoise(x * frequency, z * frequency, seed + octave * 1013) * amplitude
    normalizer += amplitude
    amplitude *= 0.5
    frequency *= 2.05
  }
  return value / normalizer
}

function mixColor(a, b, factor) {
  return [
    lerp(a[0], b[0], factor),
    lerp(a[1], b[1], factor),
    lerp(a[2], b[2], factor),
  ]
}

function tintColor(color, amount) {
  const target = amount >= 0 ? [1, 1, 1] : [0, 0, 0]
  return mixColor(color, target, Math.abs(amount))
}

function pushVertex(target, position, normal, color) {
  target.push(...position, ...normal, ...color)
}

function pushTriangle(target, a, b, c, normal, color) {
  pushVertex(target, a, normal, color)
  pushVertex(target, b, normal, color)
  pushVertex(target, c, normal, color)
}

function pushQuad(target, a, b, c, d, normal, color) {
  pushTriangle(target, a, b, c, normal, color)
  pushTriangle(target, a, c, d, normal, color)
}

function calculateNormal(a, b, c) {
  const ux = b[0] - a[0]
  const uy = b[1] - a[1]
  const uz = b[2] - a[2]
  const vx = c[0] - a[0]
  const vy = c[1] - a[1]
  const vz = c[2] - a[2]
  let nx = uy * vz - uz * vy
  let ny = uz * vx - ux * vz
  let nz = ux * vy - uy * vx
  const length = Math.hypot(nx, ny, nz) || 1
  nx /= length
  ny /= length
  nz /= length
  if (ny < 0) return [-nx, -ny, -nz]
  return [nx, ny, nz]
}

function pushBox(target, x, y, z, width, height, depth, color, rotation = 0, roofColor = null) {
  const halfWidth = width / 2
  const halfDepth = depth / 2
  const cos = Math.cos(rotation)
  const sin = Math.sin(rotation)
  const transform = (localX, localY, localZ) => [
    x + localX * cos - localZ * sin,
    y + localY,
    z + localX * sin + localZ * cos,
  ]
  const rotateNormal = (nx, ny, nz) => [nx * cos - nz * sin, ny, nx * sin + nz * cos]
  const p000 = transform(-halfWidth, 0, -halfDepth)
  const p100 = transform(halfWidth, 0, -halfDepth)
  const p110 = transform(halfWidth, height, -halfDepth)
  const p010 = transform(-halfWidth, height, -halfDepth)
  const p001 = transform(-halfWidth, 0, halfDepth)
  const p101 = transform(halfWidth, 0, halfDepth)
  const p111 = transform(halfWidth, height, halfDepth)
  const p011 = transform(-halfWidth, height, halfDepth)
  pushQuad(target, p000, p100, p110, p010, rotateNormal(0, 0, -1), color)
  pushQuad(target, p101, p001, p011, p111, rotateNormal(0, 0, 1), color)
  pushQuad(target, p001, p000, p010, p011, rotateNormal(-1, 0, 0), color)
  pushQuad(target, p100, p101, p111, p110, rotateNormal(1, 0, 0), color)
  pushQuad(target, p010, p110, p111, p011, [0, 1, 0], roofColor || color)
}

function pushGableRoof(target, x, y, z, width, depth, roofHeight, color, rotation = 0) {
  const halfWidth = width / 2
  const halfDepth = depth / 2
  const cos = Math.cos(rotation)
  const sin = Math.sin(rotation)
  const transform = (localX, localY, localZ) => [
    x + localX * cos - localZ * sin,
    y + localY,
    z + localX * sin + localZ * cos,
  ]
  const a = transform(-halfWidth, 0, -halfDepth)
  const b = transform(halfWidth, 0, -halfDepth)
  const c = transform(halfWidth, 0, halfDepth)
  const d = transform(-halfWidth, 0, halfDepth)
  const e = transform(0, roofHeight, -halfDepth)
  const f = transform(0, roofHeight, halfDepth)
  pushTriangle(target, a, b, e, calculateNormal(a, b, e), color)
  pushTriangle(target, c, d, f, calculateNormal(c, d, f), color)
  pushQuad(target, a, e, f, d, calculateNormal(a, e, f), color)
  pushQuad(target, b, c, f, e, calculateNormal(b, c, f), color)
}

function pushTree(target, x, y, z, scale, random) {
  const trunkHeight = scale * range(random, 2.6, 3.5)
  pushBox(target, x, y, z, 0.45 * scale, trunkHeight, 0.45 * scale, COLORS.treeTrunk)
  const canopy = mixColor(COLORS.treeCanopy, COLORS.treeCanopyLight, random())
  pushBox(target, x, y + trunkHeight * 0.72, z, 2.8 * scale, 2.5 * scale, 2.8 * scale, canopy, random() * Math.PI)
  if (scale > 0.75) {
    pushBox(target, x + 0.45 * scale, y + trunkHeight * 1.05, z - 0.25 * scale, 2.1 * scale, 2.0 * scale, 2.1 * scale, tintColor(canopy, 0.05), random() * Math.PI)
  }
}

function pushStreetLight(target, x, y, z, rotation = 0) {

  const armZ = z + Math.sin(rotation) * 0.7

}

function pushVehicle(target, x, y, z, rotation, random) {
  const color = pick(random, [COLORS.vehicleWhite, COLORS.vehicleRed, COLORS.vehicleBlue, [0.12, 0.13, 0.14]])

}

    return riverOffset

      + Math.sin((x / extent) * Math.PI * 4.7 - phase * 0.7) * extent * 0.018
  }
