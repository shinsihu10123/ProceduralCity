          if (isRoadRow) pushBox(vertices, x, groundY + 0.105, z, 1.55, 0.015, 0.09, COLORS.roadMark)
        }
        continue
      }

      if (random() > settings.density) continue

      const normalizedDistance = Math.min(1, Math.hypot(x, z) / (extent * 0.58))
      const centerFactor = 1 - normalizedDistance
      const zone = chooseZone(normalizedDistance, random)
      const height = buildingHeight(zone, centerFactor, random)
      const building = {
        id: buildings.length,
        x,
        y: groundY,
        z,
        width: range(random, 3.7, 5.65),
        depth: range(random, 3.7, 5.65),
        height,
        rotation: random() > 0.88 ? Math.PI / 2 : 0,
        zone,
      }
      buildings.push(building)
      pushBox(
        vertices,
        building.x,
        building.y + 0.12,
        building.z,
        building.width,
        building.height,
        building.depth,
        COLORS[zone],
        building.rotation,
        COLORS.roof,
      )

      if (building.height > 13 && random() > 0.45) {
        pushBox(
          vertices,
          building.x,
          building.y + building.height + 0.12,
          building.z,
          building.width * 0.42,
          0.65,
          building.depth * 0.42,
          COLORS.roof,
          building.rotation,
        )
      }
    }
  }

  const totalHeight = buildings.reduce((sum, building) => sum + building.height, 0)
  const population = Math.round(buildings.reduce((sum, building) => {
    const factor = building.zone === 'residential' ? 2.65 : 0.38
    return sum + building.width * building.depth * building.height * factor
  }, 0))

  const metrics = {
    buildings: buildings.length,
    population,
    averageHeight: buildings.length ? totalHeight / buildings.length : 0,
    developedRatio: buildings.length / (settings.gridSize * settings.gridSize),
  }

  return { settings, extent, buildings, roads, metrics, vertices: new Float32Array(vertices) }
}

function uploadCity(city) {
  gl.bindBuffer(gl.ARRAY_BUFFER, vertexBuffer)
  gl.bufferData(gl.ARRAY_BUFFER, city.vertices, gl.STATIC_DRAW)
  vertexCount = city.vertices.length / 9

  const stride = 9 * Float32Array.BYTES_PER_ELEMENT
  gl.enableVertexAttribArray(locations.position)
  gl.vertexAttribPointer(locations.position, 3, gl.FLOAT, false, stride, 0)
  gl.enableVertexAttribArray(locations.normal)
  gl.vertexAttribPointer(locations.normal, 3, gl.FLOAT, false, stride, 3 * Float32Array.BYTES_PER_ELEMENT)
  gl.enableVertexAttribArray(locations.color)
  gl.vertexAttribPointer(locations.color, 3, gl.FLOAT, false, stride, 6 * Float32Array.BYTES_PER_ELEMENT)
}

function updateMetrics(city) {
  ui.metricBuildings.textContent = city.metrics.buildings.toLocaleString('ko-KR')
  ui.metricPopulation.textContent = city.metrics.population.toLocaleString('ko-KR')
  ui.metricHeight.textContent = `${city.metrics.averageHeight.toFixed(1)} m`
  ui.metricRatio.textContent = `${Math.round(city.metrics.developedRatio * 100)}%`
}

function readSettings() {
  return {
    seed: ui.seed.value.trim() || 'city',
    gridSize: Number(ui.gridSize.value),
    density: Number(ui.density.value) / 100,
    roughness: Number(ui.roughness.value),
    roadSpacing: Number(ui.roadSpacing.value),
  }
}

function rebuildCity() {
  ui.loading.classList.add('visible')
  requestAnimationFrame(() => {
    currentCity = generateCity(readSettings())
    uploadCity(currentCity)
    updateMetrics(currentCity)
    camera.target = [0, 5, 0]
    camera.distance = Math.max(95, currentCity.extent * 0.95)
    ui.loading.classList.remove('visible')
  })
}

function updateLabels() {
  ui.gridSizeValue.textContent = `${ui.gridSize.value} × ${ui.gridSize.value}`
  ui.densityValue.textContent = `${ui.density.value}%`
  ui.roughnessValue.textContent = Number(ui.roughness.value).toFixed(1)
  ui.roadSpacingValue.textContent = `${ui.roadSpacing.value}칸`
}

for (const control of [ui.gridSize, ui.density, ui.roughness, ui.roadSpacing]) {
  control.addEventListener('input', updateLabels)
  control.addEventListener('change', rebuildCity)
}
ui.seed.addEventListener('keydown', (event) => {
  if (event.key === 'Enter') rebuildCity()
})
ui.generate.addEventListener('click', rebuildCity)
ui.export.addEventListener('click', () => {
  if (!currentCity) return
  const exportData = {
    schemaVersion: 1,
    generatedAt: new Date().toISOString(),
    settings: currentCity.settings,
    metrics: currentCity.metrics,
    buildings: currentCity.buildings,
    roads: currentCity.roads,
  }
  const blob = new Blob([JSON.stringify(exportData, null, 2)], { type: 'application/json' })
  const url = URL.createObjectURL(blob)
  const anchor = document.createElement('a')
  anchor.href = url
  anchor.download = `procedural-city-${currentCity.settings.seed.replace(/[^a-z0-9-_]+/gi, '-')}.json`
  anchor.click()
  URL.revokeObjectURL(url)
})
