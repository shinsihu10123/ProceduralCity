  const jobFloorArea = buildings.reduce((sum, building) => {
    if (building.zone === 'commercial') return sum + building.floorArea / 22
    if (building.zone === 'industrial') return sum + building.floorArea / 48
    if (building.zone === 'civic') return sum + building.floorArea / 34
    return sum
  }, 0)
  const population = Math.round(residentialFloorArea / 34)
  const greenRatio = parks.length / Math.max(buildableLots, 1)

  const metrics = {
    buildings: buildings.length,
    population,
    jobs: Math.round(jobFloorArea),
    averageHeight: buildings.length ? totalHeight / buildings.length : 0,
    developedRatio: developedLots / Math.max(buildableLots, 1),
    greenRatio,
  }

  return {
    schemaVersion: 2,
    settings,
    extent,
    districts: plan,
    buildings,
    roads,
    parks,
    parcels,
    metrics,
    vertices: new Float32Array(vertices),
  }
}

function uploadCity(city) {
  gl.bindBuffer(gl.ARRAY_BUFFER, vertexBuffer)
  gl.bufferData(gl.ARRAY_BUFFER, city.vertices, gl.STATIC_DRAW)
  vertexCount = city.vertices.length / 9

  const stride = 9 * Float32Array.BYTES_PER_ELEMENT
  gl.enableVertexAttribArray(locations.position)
  gl.vertexAttribPointer(locations.position, 3, gl.FLOAT, false, stride, 0)
  gl.enableVertexAttribArray(locations.normal)
  gl.vertexAttribPointer(locations.normal, 3, gl.FLOAT, false, stride, 3 * Float32Array.BYTES_PER_ELEMENT)
  gl.enableVertexAttribArray(locations.color)
  gl.vertexAttribPointer(locations.color, 3, gl.FLOAT, false, stride, 6 * Float32Array.BYTES_PER_ELEMENT)
}

const camera = {
  yaw: Math.PI * 0.25,
  pitch: 0.72,
  distance: 145,
  target: [0, 5, 0],
}

let dragging = false
let panning = false
let previousPointer = [0, 0]

canvas.addEventListener('pointerdown', (event) => {
  dragging = true
  panning = event.shiftKey || event.button === 2
  previousPointer = [event.clientX, event.clientY]
  canvas.setPointerCapture(event.pointerId)
})

canvas.addEventListener('pointermove', (event) => {
  if (!dragging) return
  const dx = event.clientX - previousPointer[0]
  const dy = event.clientY - previousPointer[1]
  previousPointer = [event.clientX, event.clientY]

  if (panning) {
    const panScale = camera.distance * 0.0018
    const right = [Math.cos(camera.yaw), 0, -Math.sin(camera.yaw)]
    const forward = [Math.sin(camera.yaw), 0, Math.cos(camera.yaw)]
    camera.target[0] -= right[0] * dx * panScale + forward[0] * dy * panScale
    camera.target[2] -= right[2] * dx * panScale + forward[2] * dy * panScale
  } else {
    camera.yaw -= dx * 0.006
    camera.pitch = Math.min(1.48, Math.max(0.12, camera.pitch + dy * 0.005))
  }
})

canvas.addEventListener('pointerup', (event) => {
  dragging = false
  canvas.releasePointerCapture(event.pointerId)
})
canvas.addEventListener('pointercancel', () => { dragging = false })
canvas.addEventListener('contextmenu', (event) => event.preventDefault())
canvas.addEventListener('wheel', (event) => {
  event.preventDefault()
  camera.distance = Math.min(520, Math.max(20, camera.distance * Math.exp(event.deltaY * 0.001)))
}, { passive: false })

function mat4Perspective(out, fovy, aspect, near, far) {
  const f = 1 / Math.tan(fovy / 2)
  out.fill(0)
  out[0] = f / aspect
  out[5] = f
  out[10] = (far + near) / (near - far)
  out[11] = -1
  out[14] = (2 * far * near) / (near - far)
  return out
}

function normalize3(vector) {
  const length = Math.hypot(vector[0], vector[1], vector[2]) || 1
  return [vector[0] / length, vector[1] / length, vector[2] / length]
}

function cross3(a, b) {
  return [
    a[1] * b[2] - a[2] * b[1],
    a[2] * b[0] - a[0] * b[2],
    a[0] * b[1] - a[1] * b[0],
  ]
}

function dot3(a, b) {
  return a[0] * b[0] + a[1] * b[1] + a[2] * b[2]
}

function mat4LookAt(out, eye, target, up) {
  const zAxis = normalize3([eye[0] - target[0], eye[1] - target[1], eye[2] - target[2]])
  const xAxis = normalize3(cross3(up, zAxis))
  const yAxis = cross3(zAxis, xAxis)

  out[0] = xAxis[0]; out[1] = yAxis[0]; out[2] = zAxis[0]; out[3] = 0
  out[4] = xAxis[1]; out[5] = yAxis[1]; out[6] = zAxis[1]; out[7] = 0
  out[8] = xAxis[2]; out[9] = yAxis[2]; out[10] = zAxis[2]; out[11] = 0
  out[12] = -dot3(xAxis, eye); out[13] = -dot3(yAxis, eye); out[14] = -dot3(zAxis, eye); out[15] = 1
  return out
}

function mat4Multiply(out, a, b) {
  const result = new Float32Array(16)
  for (let column = 0; column < 4; column += 1) {
    for (let row = 0; row < 4; row += 1) {
      result[column * 4 + row] =
        a[0 * 4 + row] * b[column * 4 + 0] +
        a[1 * 4 + row] * b[column * 4 + 1] +
        a[2 * 4 + row] * b[column * 4 + 2] +
        a[3 * 4 + row] * b[column * 4 + 3]
    }
  }
  out.set(result)
  return out
}

const projection = new Float32Array(16)
const view = new Float32Array(16)
const viewProjection = new Float32Array(16)

function resizeCanvas() {
  const pixelRatio = Math.min(window.devicePixelRatio || 1, 2)
  const width = Math.max(1, Math.floor(canvas.clientWidth * pixelRatio))
  const height = Math.max(1, Math.floor(canvas.clientHeight * pixelRatio))
  if (canvas.width !== width || canvas.height !== height) {
    canvas.width = width
    canvas.height = height
  }
  gl.viewport(0, 0, width, height)
}

function render() {
  resizeCanvas()
  gl.enable(gl.DEPTH_TEST)
  gl.disable(gl.CULL_FACE)
  gl.clearColor(0.63, 0.75, 0.82, 1)
  gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT)

  const cosPitch = Math.cos(camera.pitch)
  const eye = [
    camera.target[0] + Math.sin(camera.yaw) * cosPitch * camera.distance,
    camera.target[1] + Math.sin(camera.pitch) * camera.distance,
    camera.target[2] + Math.cos(camera.yaw) * cosPitch * camera.distance,
  ]

  mat4Perspective(projection, Math.PI / 4, canvas.width / canvas.height, 0.1, 1400)
  mat4LookAt(view, eye, camera.target, [0, 1, 0])
  mat4Multiply(viewProjection, projection, view)

  gl.useProgram(program)
  gl.uniformMatrix4fv(locations.viewProjection, false, viewProjection)
  gl.uniform3f(locations.lightDirection, 0.55, 0.92, 0.36)
  gl.uniform3f(locations.fogColor, 0.63, 0.75, 0.82)
  gl.uniform1f(locations.fogNear, currentCity ? currentCity.extent * 0.95 : 160)
  gl.uniform1f(locations.fogFar, currentCity ? currentCity.extent * 2.4 : 360)
  gl.bindBuffer(gl.ARRAY_BUFFER, vertexBuffer)
  gl.drawArrays(gl.TRIANGLES, 0, vertexCount)
  requestAnimationFrame(render)
}
