
const v052RailBuffer = gl.createBuffer()
let v052RailCity = null
let v052RailVertexCount = 0
let v052RailProgress = 0.12
let v052RailDirection = 1
let v052RailDwell = 0
let v052RailStations = [0.18, 0.5, 0.82]
let v052RailLastFrame = performance.now()

function v052RailBindGeometry(buffer) {
  gl.bindBuffer(gl.ARRAY_BUFFER, buffer)
  const stride = 9 * Float32Array.BYTES_PER_ELEMENT
  gl.enableVertexAttribArray(locations.position)
  gl.vertexAttribPointer(locations.position, 3, gl.FLOAT, false, stride, 0)
  gl.enableVertexAttribArray(locations.normal)
  gl.vertexAttribPointer(locations.normal, 3, gl.FLOAT, false, stride, 3 * Float32Array.BYTES_PER_ELEMENT)
  gl.enableVertexAttribArray(locations.color)
  gl.vertexAttribPointer(locations.color, 3, gl.FLOAT, false, stride, 6 * Float32Array.BYTES_PER_ELEMENT)
}

function v052RailReset(city) {
  v052RailCity = city
  v052RailProgress = 0.12
  v052RailDirection = 1
  v052RailDwell = 0
  const stations = city?.infrastructure?.stations || []
  if (stations.length >= 2) {
    const extent = Math.max(city.extent || 1, 1)
    v052RailStations = stations
      .map((station) => clamp((station.z / extent) + 0.5, 0.08, 0.92))
      .sort((a, b) => a - b)
  } else {
    v052RailStations = [0.18, 0.5, 0.82]
  }
}

function v052RailNearestStation(progress) {
  let nearest = null
  let distance = Infinity
  for (const station of v052RailStations) {
    const candidate = Math.abs(progress - station)
    if (candidate < distance) {
      nearest = station
      distance = candidate
    }
  }
  return { station: nearest, distance }
}

function v052RailPushCar(vertices, x, y, z, color, lead = false) {
  pushBox(vertices, x, y, z, 3.15, 2.25, 12.6, color, 0)
  pushBox(vertices, x, y + 1.45, z, 3.02, 0.92, 10.8, [0.12, 0.25, 0.32], 0)
  pushBox(vertices, x, y + 2.46, z, 2.24, 0.16, 7.4, [0.70, 0.76, 0.78], 0)
  pushBox(vertices, x - 1.59, y + 0.38, z, 0.08, 0.82, 8.8, [0.08, 0.16, 0.21], 0)
  pushBox(vertices, x + 1.59, y + 0.38, z, 0.08, 0.82, 8.8, [0.08, 0.16, 0.21], 0)
  if (lead) {
    pushBox(vertices, x, y + 0.30, z + 6.34 * v052RailDirection, 1.75, 0.18, 0.10, [1.0, 0.84, 0.42], 0)
  }
}

function v052RailBuild(city, deltaSeconds) {
  if (city !== v052RailCity) v052RailReset(city)
  if (!city) {
    v052RailVertexCount = 0
    return
  }

  const routeLength = Math.max((city.extent || 600) * 0.92, 160)
  if (v052RailDwell > 0) {
    v052RailDwell = Math.max(0, v052RailDwell - deltaSeconds)
  } else {
    const previous = v052RailProgress
    v052RailProgress += deltaSeconds * 14.5 * v052RailDirection / routeLength
    if (v052RailProgress >= 0.96) {
      v052RailProgress = 0.96
      v052RailDirection = -1
      v052RailDwell = 1.8
    } else if (v052RailProgress <= 0.04) {
      v052RailProgress = 0.04
      v052RailDirection = 1
      v052RailDwell = 1.8
    } else {
      const nearest = v052RailNearestStation(v052RailProgress)
      const crossed = nearest.station !== null
        && ((v052RailDirection > 0 && previous < nearest.station && v052RailProgress >= nearest.station)
          || (v052RailDirection < 0 && previous > nearest.station && v052RailProgress <= nearest.station))
      if (crossed) {
        v052RailProgress = nearest.station
        v052RailDwell = 2.4
      }
    }
  }

  const vertices = []
  const railX = city.centers?.cbd?.x || 0
  const baseY = (city.infrastructure?.stations?.[0]?.y ?? 1.8) + 0.78
  const trainZ = -routeLength / 2 + v052RailProgress * routeLength
  const color = [0.18, 0.63, 0.79]
  const spacing = 13.8
  for (let car = -1; car <= 1; car += 1) {
    const z = trainZ + car * spacing * -v052RailDirection
    v052RailPushCar(vertices, railX, baseY, z, color, car === -v052RailDirection)
  }

  const data = new Float32Array(vertices)
  gl.bindBuffer(gl.ARRAY_BUFFER, v052RailBuffer)
  gl.bufferData(gl.ARRAY_BUFFER, data, gl.DYNAMIC_DRAW)
  v052RailVertexCount = data.length / 9
}

function v052RailRender(frameTime = performance.now()) {
  const deltaSeconds = Math.min(0.05, Math.max(0, (frameTime - v052RailLastFrame) / 1000))
  v052RailLastFrame = frameTime
  if (currentCity && program && v052RailBuffer) {
    v052RailBuild(currentCity, deltaSeconds)
    if (v052RailVertexCount > 0) {
      gl.enable(gl.DEPTH_TEST)
      gl.useProgram(program)
      v052RailBindGeometry(v052RailBuffer)
      gl.drawArrays(gl.TRIANGLES, 0, v052RailVertexCount)
    }
  }
  requestAnimationFrame(v052RailRender)
}

window.ProceduralCityV052Rail = {
  getState: () => ({
    progress: v052RailProgress,
    direction: v052RailDirection,
    dwelling: v052RailDwell > 0,
    dwellSeconds: v052RailDwell,
    stations: [...v052RailStations],
    cars: 3,
  }),
}

setTimeout(() => requestAnimationFrame(v052RailRender), 0)
